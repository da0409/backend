cmake_minimum_required(VERSION 3.16)
project(timecapsule_geo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(JNI REQUIRED COMPONENTS JVM)
add_library(timecapsule_geo SHARED geo.cpp jni.cpp)
target_include_directories(timecapsule_geo PRIVATE ${JNI_INCLUDE_DIRS})
target_compile_options(timecapsule_geo PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_executable(geo_test geo.cpp geo_test.cpp)
add_test(NAME geo_boundaries COMMAND geo_test)
